    <script>
        const API_BASE = 'http://localhost:8080/api';
        let allData = { transactions: [], analytics: {} };
        let allCharts = {};
        let filters = { search: '', category: '', type: '', hmmState: '', page: 1, pageSize: 10 };

        async function loadData() {
            try {
                const [txRes, analyticsRes] = await Promise.all([
                    fetch(`${API_BASE}/transactions`),
                    fetch(`${API_BASE}/analytics`)
                ]);
                
                if (!txRes.ok || !analyticsRes.ok) throw new Error('API Error');
                
                allData.transactions = await txRes.json();
                allData.analytics = await analyticsRes.json();
                
                updateUI();
            } catch (error) {
                console.error('Load error:', error);
            }
        }

        function updateUI() {
            updateCardsSummary();
            updateCharts();
            updateInsights();
            updateHMM();
            updateHMMInsights();
            updateHMMAlerts();
            updateHMMDistributionChart();
            updateAnomalySummary();
            updateAnomalies();
            updateTransactionTable();
            populateFilters();
        }

        function updateCardsSummary() {
            const { totalIncome = 0, totalExpenses = 0, accountBalance = 0 } = allData.analytics;
            const net = totalIncome - totalExpenses;
            const rate = totalIncome > 0 ? ((net / totalIncome) * 100) : 0;
            
            document.getElementById('totalIncome').textContent = `â‚¹${format(totalIncome)}`;
            document.getElementById('totalExpenses').textContent = `â‚¹${format(totalExpenses)}`;
            document.getElementById('netSavings').textContent = `â‚¹${format(net)}`;
            document.getElementById('savingsRate').textContent = `${rate.toFixed(1)}%`;
            document.getElementById('totalBalance').textContent = `â‚¹${format(accountBalance)}`;
        }

        function updateCharts() {
            const cats = allData.analytics.categoryBreakdown || {};
            const ctx1 = document.getElementById('categoryChart').getContext('2d');
            if (allCharts.cat) allCharts.cat.destroy();
            allCharts.cat = new Chart(ctx1, {
                type: 'doughnut',
                data: {
                    labels: Object.keys(cats),
                    datasets: [{
                        data: Object.values(cats),
                        backgroundColor: ['#22c55e', '#ef4444', '#3b82f6', '#a855f7', '#ec4899', '#f59e0b', '#06b6d4'],
                        borderColor: '#0f172a',
                        borderWidth: 2
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { labels: { color: '#93c5fd' } } } }
            });

            const ctx2 = document.getElementById('trendChart').getContext('2d');
            if (allCharts.trend) allCharts.trend.destroy();
            allCharts.trend = new Chart(ctx2, {
                type: 'line',
                data: {
                    labels: ['Week 1', 'Week 2', 'Week 3', 'Week 4'],
                    datasets: [{
                        label: 'Spending',
                        data: [2000, 2500, 3000, 2200],
                        borderColor: '#3b82f6',
                        backgroundColor: 'rgba(59, 130, 246, 0.1)',
                        tension: 0.4
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { labels: { color: '#93c5fd' } } } }
            });
        }

        function updateHMM() {
            const states = allData.analytics.hmmStates || {};
            const entries = Object.entries(states);
            let low = 0, normal = 0, high = 0;
            entries.forEach(([_, s]) => {
                if (s.includes('low')) low++;
                else if (s.includes('high')) high++;
                else normal++;
            });
            const total = entries.length || 1;
            document.getElementById('stateLowCount').textContent = low;
            document.getElementById('stateLowPercent').textContent = `${((low/total)*100).toFixed(0)}%`;
            document.getElementById('stateNormalCount').textContent = normal;
            document.getElementById('stateNormalPercent').textContent = `${((normal/total)*100).toFixed(0)}%`;
            document.getElementById('stateHighCount').textContent = high;
            document.getElementById('stateHighPercent').textContent = `${((high/total)*100).toFixed(0)}%`;

            const ctx = document.getElementById('hmmStateChart').getContext('2d');
            if (allCharts.hmm) allCharts.hmm.destroy();
            allCharts.hmm = new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: ['Low', 'Normal', 'High'],
                    datasets: [{ data: [low, normal, high], backgroundColor: ['#22c55e', '#eab308', '#ef4444'] }]
                },
                options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } } }
            });
        }

        function updateInsights() {
            const cats = allData.analytics.categoryBreakdown || {};
            const top = Object.entries(cats).sort((a, b) => b[1] - a[1])[0];
            if (top) {
                const total = Object.values(cats).reduce((a, b) => a + b, 0);
                document.getElementById('topCategory').textContent = top[0];
                document.getElementById('topCategoryPercent').textContent = `${((top[1]/total)*100).toFixed(1)}%`;
            }

            const merchants = {};
            allData.transactions.forEach(t => { merchants[t.merchant_person] = (merchants[t.merchant_person] || 0) + 1; });
            const topM = Object.entries(merchants).sort((a, b) => b[1] - a[1])[0];
            if (topM) {
                document.getElementById('topMerchant').textContent = topM[0];
                document.getElementById('topMerchantCount').textContent = `${topM[1]} txns`;
            }

            const highest = allData.transactions.reduce((max, t) => parseFloat(t.amount) > parseFloat(max.amount) ? t : max, allData.transactions[0] || {});
            document.getElementById('unusualAmount').textContent = `â‚¹${format(highest.amount || 0)}`;
            document.getElementById('recurringCount').textContent = allData.transactions.filter(t => t.is_recurring).length;
        }

        function updateAnomalies() {
            const anom = allData.analytics.anomalies || [];
            const cont = document.getElementById('anomaliesContainer');
            cont.innerHTML = anom.length ? anom.map(a => `<div class="card-hover p-4 rounded-lg border-l-4 border-red-500"><span class="anomaly-badge">âš  ANOMALY</span> ${a.description} - â‚¹${format(a.amount)}</div>`).join('') : '<div class="card-hover p-4 rounded-lg text-center text-gray-400">âœ“ No anomalies detected</div>';
        }

        function updateHMMInsights() {
            const container = document.getElementById('hmmInsightsContainer');
            if (!container) return;

            // Generate insights based on HMM data
            const insights = [];
            const states = allData.analytics.hmmStates || {};
            const entries = Object.entries(states);
            
            if (entries.length === 0) {
                container.innerHTML = '<div class="text-gray-400 text-sm">No transaction data available yet.</div>';
                return;
            }

            // State dominance insight
            const stateCounts = {};
            entries.forEach(([_, state]) => {
                const label = (state || 'normal').toLowerCase().split('_')[0];
                stateCounts[label] = (stateCounts[label] || 0) + 1;
            });
            
            const dominant = Object.entries(stateCounts).sort((a, b) => b[1] - a[1])[0];
            if (dominant) {
                const pct = ((dominant[1] / entries.length) * 100).toFixed(0);
                const stateEmoji = dominant[0] === 'low' ? 'ðŸŸ¢' : dominant[0] === 'high' ? 'ðŸ”´' : 'ðŸŸ¡';
                insights.push({
                    icon: 'ðŸ“Š',
                    title: 'Dominant Pattern',
                    text: `${stateEmoji} You spend in <strong>${dominant[0]} spending</strong> mode <strong>${pct}%</strong> of the time.`
                });
            }

            // Trend insight
            const txCount = (allData.transactions || []).length;
            if (txCount > 0) {
                const avgAmount = allData.transactions.reduce((sum, t) => sum + parseFloat(t.amount || 0), 0) / txCount;
                insights.push({
                    icon: 'ðŸ’¡',
                    title: 'Transaction Average',
                    text: `Typical transaction is around <strong>â‚¹${format(avgAmount)}</strong>.`
                });
            }

            // Volatility insight
            if (allData.analytics.categoryBreakdown) {
                const cats = Object.values(allData.analytics.categoryBreakdown).filter(v => v > 0);
                if (cats.length > 1) {
                    const variance = cats.map(v => Math.abs(v - cats.reduce((a, b) => a + b) / cats.length)).reduce((a, b) => a + b) / cats.length;
                    const volatility = variance / (cats.reduce((a, b) => a + b) / cats.length);
                    if (volatility > 0.5) {
                        insights.push({
                            icon: 'ðŸ“ˆ',
                            title: 'Spending Volatility',
                            text: '<strong>High variance</strong> across categories. Consider establishing consistent budget allocations.'
                        });
                    } else {
                        insights.push({
                            icon: 'âœ…',
                            title: 'Spending Consistency',
                            text: 'Your spending is <strong>well-balanced</strong> across categories.'
                        });
                    }
                }
            }

            // Anomaly count
            const anomalies = allData.analytics.anomalies || [];
            if (anomalies.length > 0) {
                insights.push({
                    icon: 'âš ï¸',
                    title: 'Anomalies Detected',
                    text: `Found <strong>${anomalies.length}</strong> unusual transaction(s). Check the anomalies section below.`
                });
            } else {
                insights.push({
                    icon: 'âœ¨',
                    title: 'Normal Behavior',
                    text: 'No unusual spending patterns detected. Keep up the good habits!'
                });
            }

            container.innerHTML = insights.map(insight => `
                <div class="p-4 rounded-lg bg-slate-800/30 border border-blue-500/20 hover:border-blue-500/50 transition">
                    <div class="flex items-start gap-3">
                        <span class="text-2xl">${insight.icon}</span>
                        <div>
                            <div class="font-semibold text-white text-sm">${insight.title}</div>
                            <div class="text-xs text-gray-300 mt-1">${insight.text}</div>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        function updateHMMAlerts() {
            const container = document.getElementById('hmmAlertsContainer');
            if (!container) return;

            const alerts = [];

            // Check for risky states
            const states = allData.analytics.hmmStates || {};
            const highCount = Object.values(states).filter(s => (s || '').toLowerCase().includes('high')).length;
            const riskCount = Object.values(states).filter(s => (s || '').toLowerCase().includes('risky')).length;

            if (riskCount > 3) {
                alerts.push({
                    level: 'high',
                    icon: 'ðŸ”´',
                    title: 'Risky Spending',
                    text: `${riskCount} days of risky spending detected.`
                });
            }

            if (highCount > 5) {
                alerts.push({
                    level: 'medium',
                    icon: 'ðŸŸ ',
                    title: 'High Spending Days',
                    text: `${highCount} days with high spending identified.`
                });
            }

            // Check anomalies
            const anomalies = allData.analytics.anomalies || [];
            if (anomalies.length > 2) {
                alerts.push({
                    level: 'medium',
                    icon: 'âš ï¸',
                    title: `Multiple Anomalies (${anomalies.length})`,
                    text: 'Review unusual transactions in detail.'
                });
            }

            // Check spending rate
            const totalIncome = allData.analytics.totalIncome || 1;
            const totalExpenses = allData.analytics.totalExpenses || 0;
            const savingsRate = ((totalIncome - totalExpenses) / totalIncome) * 100;
            if (savingsRate < 10) {
                alerts.push({
                    level: 'high',
                    icon: 'ðŸ’¸',
                    title: 'Low Savings Rate',
                    text: `Saving only ${savingsRate.toFixed(1)}%. Aim to save at least 20%.`
                });
            }

            if (alerts.length === 0) {
                container.innerHTML = '<div class="text-gray-400 text-sm">âœ“ No critical alerts</div>';
            } else {
                container.innerHTML = alerts.map(alert => `
                    <div class="p-3 rounded-lg ${alert.level === 'high' ? 'bg-red-500/10 border border-red-500/30' : 'bg-yellow-500/10 border border-yellow-500/30'}">
                        <div class="flex items-start gap-2">
                            <span class="text-lg">${alert.icon}</span>
                            <div>
                                <div class="font-semibold text-white text-xs">${alert.title}</div>
                                <div class="text-xs text-gray-300 mt-1">${alert.text}</div>
                            </div>
                        </div>
                    </div>
                `).join('');
            }
        }

        function updateHMMDistributionChart() {
            const states = allData.analytics.hmmStates || {};
            const entries = Object.entries(states);
            
            let low = 0, normal = 0, high = 0, risky = 0, veryHigh = 0;
            entries.forEach(([_, s]) => {
                const label = (s || '').toLowerCase();
                if (label.includes('low')) low++;
                else if (label.includes('very_high') || label.includes('very high')) veryHigh++;
                else if (label.includes('high')) high++;
                else if (label.includes('risky')) risky++;
                else normal++;
            });

            const ctx = document.getElementById('hmmDistributionChart').getContext('2d');
            if (allCharts.hmmDist) allCharts.hmmDist.destroy();
            allCharts.hmmDist = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['Low Spending', 'Normal', 'High Spending', 'Risky', 'Very High'],
                    datasets: [{
                        data: [low, normal, high, risky, veryHigh],
                        backgroundColor: ['#22c55e', '#eab308', '#ef4444', '#ec4899', '#dc2626'],
                        borderColor: '#0f172a',
                        borderWidth: 2
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: {
                            position: 'bottom',
                            labels: { color: '#93c5fd', padding: 15 }
                        }
                    }
                }
            });
        }

        function updateAnomalySummary() {
            const container = document.getElementById('anomalySummaryContainer');
            if (!container) return;

            const anomalies = allData.analytics.anomalies || [];
            const totalTxns = (allData.transactions || []).length;
            const anomalyRate = totalTxns > 0 ? ((anomalies.length / totalTxns) * 100).toFixed(1) : 0;

            const summary = [];
            summary.push({
                label: 'Total Anomalies',
                value: anomalies.length,
                color: anomalies.length > 0 ? 'text-red-400' : 'text-green-400'
            });
            summary.push({
                label: 'Anomaly Rate',
                value: `${anomalyRate}%`,
                color: anomalyRate > 5 ? 'text-orange-400' : 'text-green-400'
            });
            summary.push({
                label: 'Highest Anomaly',
                value: anomalies.length > 0 ? `â‚¹${format(Math.max(...anomalies.map(a => a.amount || 0)))}` : 'â€”',
                color: 'text-blue-400'
            });

            container.innerHTML = summary.map(item => `
                <div class="flex justify-between items-center p-3 rounded-lg bg-slate-800/30">
                    <span class="text-xs text-gray-400">${item.label}</span>
                    <span class="text-lg font-bold ${item.color}">${item.value}</span>
                </div>
            `).join('');
        }


        function updateTransactionTable() {
            let txns = allData.transactions;
            if (filters.search) txns = txns.filter(t => t.merchant_person.toLowerCase().includes(filters.search.toLowerCase()));
            if (filters.category) txns = txns.filter(t => t.category === filters.category);
            if (filters.type) txns = txns.filter(t => t.type === filters.type);
            if (filters.hmmState) txns = txns.filter(t => t.hmm_state && t.hmm_state.includes(filters.hmmState));

            const start = (filters.page - 1) * filters.pageSize;
            const paged = txns.slice(start, start + filters.pageSize);

            document.getElementById('transactionsTableBody').innerHTML = paged.map(t => `
                <tr class="border-b border-blue-500/20 hover:bg-blue-500/5">
                    <td class="px-6 py-3 text-sm">${t.transaction_date}</td>
                    <td class="px-6 py-3 text-sm font-medium text-blue-300">${t.merchant_person}</td>
                    <td class="px-6 py-3 text-sm"><span class="px-2 py-1 rounded text-xs ${t.type === 'INCOME' ? 'bg-green-500/20 text-green-300' : 'bg-red-500/20 text-red-300'}">${t.type === 'INCOME' ? 'ðŸ’° Credit' : 'ðŸ’¸ Debit'}</span></td>
                    <td class="px-6 py-3 text-sm text-right font-semibold ${t.type === 'INCOME' ? 'text-green-400' : 'text-red-400'}">${t.type === 'INCOME' ? '+' : '-'}â‚¹${format(t.amount)}</td>
                    <td class="px-6 py-3 text-sm"><span class="px-2 py-1 bg-purple-500/20 text-purple-300 rounded text-xs">${t.category || 'N/A'}</span></td>
                    <td class="px-6 py-3 text-sm">${getStateEmoji(t.hmm_state)} ${getStateReadable(t.hmm_state)}</td>
                </tr>
            `).join('');

            const total = txns.length;
            const pages = Math.ceil(total / filters.pageSize);
            document.getElementById('paginationContainer').innerHTML = [...Array(pages)].map((_, i) => `
                <button class="filter-btn ${i + 1 === filters.page ? 'active' : ''}" onclick="filters.page = ${i + 1}; updateTransactionTable();">${i + 1}</button>
            `).join('');
        }

        function getStateEmoji(state) {
            if (!state) return 'ðŸŸ¡';
            return state.includes('low') ? 'ðŸŸ¢' : state.includes('high') ? 'ðŸ”´' : 'ðŸŸ¡';
        }

        function getStateReadable(state) {
            if (!state) return 'Normal';
            return state.includes('low') ? 'Low' : state.includes('high') ? 'High' : 'Normal';
        }

        function format(n) {
            return new Intl.NumberFormat('en-IN', { minimumFractionDigits: 0, maximumFractionDigits: 2 }).format(n);
        }
