function loadAuthState() {
  try {
    const raw = localStorage.getItem(AUTH_STORAGE_KEY);
    return raw ? JSON.parse(raw) : { accessToken: null, refreshToken: null, user: null };
  } catch (err) {
    return { accessToken: null, refreshToken: null, user: null };
  }
}

function saveAuthState(nextState) {
  authState = {
    accessToken: nextState?.accessToken ?? null,
    refreshToken: nextState?.refreshToken ?? null,
    user: nextState?.user ?? null,
  };
  localStorage.setItem(AUTH_STORAGE_KEY, JSON.stringify(authState));
  syncAuthUi();
}

function clearAuthState() {
  authState = { accessToken: null, refreshToken: null, user: null };
  localStorage.removeItem(AUTH_STORAGE_KEY);
  syncAuthUi();
}

function getInitials(name) {
  if (!name) return 'FT';
  return name
    .trim()
    .split(/\s+/)
    .slice(0, 2)
    .map(part => part[0]?.toUpperCase() || '')
    .join('') || 'FT';
}

function setConnectionState(label, tone = 'var(--muted)') {
  const liveInd = document.querySelector('.live-ind');
  if (!liveInd) return;
  liveInd.innerHTML = `<div class="pulse" style="background:${tone};box-shadow:0 0 8px ${tone}"></div>${label}`;
}

function syncAuthUi() {
  const authenticated = !!authState.accessToken;
  const userName = authState.user?.fullName || authState.user?.email || 'Demo Mode';
  const initials = getInitials(userName);

  const nameEl = document.getElementById('userName');
  const initialsEl = document.getElementById('userInitials');
  const authCard = document.getElementById('authCard');
  const authSessionName = document.getElementById('authSessionName');
  const authSessionMeta = document.getElementById('authSessionMeta');
  const authClose = document.getElementById('authClose');

  if (nameEl) nameEl.textContent = userName;
  if (initialsEl) initialsEl.textContent = initials;
  if (authCard) authCard.classList.toggle('is-authenticated', authenticated);
  if (authSessionName) authSessionName.textContent = userName;
  if (authSessionMeta) {
    authSessionMeta.textContent = authenticated
      ? `${authState.user?.email || 'Authenticated'}${authState.user?.currency ? ` - Base currency ${authState.user.currency}` : ''}`
      : 'The dashboard is currently showing its built-in fallback data.';
  }
  if (authClose) authClose.textContent = authenticated ? 'Done' : 'Close';

  if (!authenticated) {
    setConnectionState('Sign In', 'var(--ember)');
  }
}

async function authRequest(path, payload) {
  const res = await fetch(`${BASE_URL}${path}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload),
  });

  if (!res.ok) {
    let message = `HTTP ${res.status}`;
    try {
      const data = await res.json();
      message = data?.message || message;
    } catch (err) {}
    throw new Error(message);
  }

  return res.status === 204 ? null : res.json();
}

async function refreshAccessToken() {
  if (!authState.refreshToken) return false;
  try {
    const next = await authRequest('/auth/refresh', { refreshToken: authState.refreshToken });
    saveAuthState({
      accessToken: next.accessToken,
      refreshToken: next.refreshToken,
      user: next.user || authState.user,
    });
    return true;
  } catch (err) {
    clearAuthState();
    return false;
  }
}

async function api(path, options = {}) {
  const { retryAuth = true, throwOnError = false, ...fetchOptions } = options;
  try {
    const isFormData = typeof FormData !== 'undefined' && fetchOptions.body instanceof FormData;
    const baseHeaders = {
      ...(authState.accessToken ? { Authorization: `Bearer ${authState.accessToken}` } : {}),
      ...(fetchOptions.headers || {}),
    };
    const res = await fetch(`${BASE_URL}${path}`, {
      ...fetchOptions,
      headers: isFormData ? baseHeaders : { 'Content-Type': 'application/json', ...baseHeaders }
    });
    if (res.status === 401 && retryAuth && authState.refreshToken) {
      const refreshed = await refreshAccessToken();
      if (refreshed) return api(path, { ...options, retryAuth: false });
    }
    if ((res.status === 401 || res.status === 403) && !authState.refreshToken) {
      clearAuthState();
      showAuthModal('login');
    }
    if (res.status === 403 && retryAuth && authState.refreshToken) {
      const refreshed = await refreshAccessToken();
      if (refreshed) return api(path, { ...options, retryAuth: false });
      clearAuthState();
      showAuthModal('login');
    }
    if (!res.ok) {
      let message = `HTTP ${res.status}`;
      try {
        const data = await res.json();
        message = data?.message || message;
      } catch (err) {}
      if (res.status === 401 || res.status === 403) {
        message = 'Session expired. Please log in again.';
      }
      throw new Error(message);
    }
    return res.status === 204 ? null : await res.json();
  } catch (err) {
    console.warn(`[FinTrack] API call failed for ${path}:`, err.message);
    if (throwOnError) throw err;
    return null;
  }
}

function signedAmount(tx) {
  const raw = Math.abs(Number(tx?.amount ?? 0));
  const type = (tx?.type ?? '').toUpperCase();
  if (type === 'EXPENSE') return -raw;
  if (type === 'TRANSFER') return 0;
  return raw;
}

function relativeDate(dateStr) {
  if (!dateStr) return '--';
  const d = new Date(dateStr);
  const now = new Date();
  const diff = Math.floor((now - d) / 86400000);
  if (diff === 0) return 'Today';
  if (diff === 1) return 'Yesterday';
  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
}

function normalizeLabel(value, fallback = 'Other') {
  return (value ?? '').toString().trim() || fallback;
}

function deriveBudgetRowsFromCategories(categories) {
  const rows = Array.isArray(categories) ? categories : [];
  const total = rows.reduce((sum, item) => sum + Number(item.totalSpent ?? item.amount ?? 0), 0);
  return rows
    .filter(item => Number(item.totalSpent ?? item.amount ?? 0) > 0)
    .slice(0, 5)
    .map(item => {
      const spent = Number(item.totalSpent ?? item.amount ?? 0);
      const share = total > 0 ? Math.round((spent / total) * 100) : 0;
      return {
        name: item.name ?? item.category ?? 'Category',
        spent,
        limit: total || spent || 1,
        percentUsed: share,
        derived: true,
      };
    });
}

function deriveRecurringCandidates(transactions) {
  const grouped = new Map();
  (Array.isArray(transactions) ? transactions : []).forEach(tx => {
    if ((tx?.type ?? '').toUpperCase() !== 'EXPENSE') return;
    const key = normalizeLabel(tx.merchant ?? tx.description, '').toUpperCase();
    if (!key) return;
    const existing = grouped.get(key) ?? {
      label: tx.merchant ?? tx.description ?? 'Transaction',
      category: tx.category ?? tx.categoryName ?? 'Uncategorized',
      count: 0,
      total: 0,
      lastDate: tx.date ?? tx.transactionDate ?? null,
    };
    existing.count += 1;
    existing.total += Math.abs(signedAmount(tx));
    const txDate = tx.date ?? tx.transactionDate ?? null;
    if (txDate && (!existing.lastDate || new Date(txDate) > new Date(existing.lastDate))) {
      existing.lastDate = txDate;
    }
    grouped.set(key, existing);
  });

  return [...grouped.values()]
    .filter(item => item.count >= 2)
    .sort((a, b) => b.count - a.count || b.total - a.total)
    .slice(0, 5);
}

function insightTone(severity) {
  const key = (severity ?? '').toUpperCase();
  if (key === 'WARNING' || key === 'HIGH' || key === 'MEDIUM') return { label: 'FLAG', cls: 'dn', style: '' };
  if (key === 'INFO') return { label: 'INFO', cls: '', style: 'color:var(--ice)' };
  return { label: 'LIVE', cls: 'up', style: '' };
}

function renderOverviewInsights(insights, dashboard) {
  const list = document.getElementById('overviewInsightsList');
  if (!list) return;

  const rows = [];
  const patterns = Array.isArray(insights?.patterns) ? insights.patterns : [];
  const recommendations = Array.isArray(insights?.recommendations) ? insights.recommendations : [];
  const anomalies = Array.isArray(insights?.anomalies) ? insights.anomalies : [];
  const budgetAlerts = Array.isArray(dashboard?.budgetAlerts) ? dashboard.budgetAlerts : [];

  patterns.slice(0, 2).forEach(item => rows.push({
    title: item.title ?? 'Pattern',
    detail: item.detail ?? '',
    severity: item.severity ?? 'INFO',
  }));
  recommendations.slice(0, 1).forEach(item => rows.push({
    title: item.title ?? 'Recommendation',
    detail: item.detail ?? '',
    severity: 'WARNING',
  }));
  anomalies.slice(0, 1).forEach(item => rows.push({
    title: item.description ?? 'Anomaly',
    detail: item.reason ?? 'Unusual activity detected.',
    severity: item.severity ?? 'WARNING',
  }));
  budgetAlerts.slice(0, 1).forEach(item => rows.push({
    title: item.name ?? 'Budget alert',
    detail: `${Number(item.percentUsed ?? 0).toFixed(0)}% of the configured budget has been used.`,
    severity: item.status === 'EXCEEDED' ? 'HIGH' : 'WARNING',
  }));

  if (!rows.length) {
    list.innerHTML = `<div class="page-loading" style="padding:12px 0">No AI insights yet<span class="blink">_</span></div>`;
    return;
  }

  list.innerHTML = rows.slice(0, 4).map(item => {
    const tone = insightTone(item.severity);
    return `
      <div class="insight-row clickable">
        <div class="insight-tag ${tone.cls}" ${tone.style ? `style="${tone.style}"` : ''}>${tone.label}</div>
        <div class="insight-text"><strong>${item.title}</strong> ${item.detail}</div>
      </div>`;
  }).join('');
}

// Animated counter (reusable)
function animCount(el, target, dur = 1800) {
  const start = performance.now();
  (function frame(now) {
    const p = Math.min((now - start) / dur, 1);
    const ease = 1 - Math.pow(1 - p, 4);
    el.textContent = fmt(ease * target);
    if (p < 1) requestAnimationFrame(frame);
  })(performance.now());
}

// Animated hero number (splits on decimal)
function animHeroNumber(intEl, decEl, target, dur = 2200) {
  const start = performance.now();
  (function frame(now) {
    const p = Math.min((now - start) / dur, 1);
    const ease = 1 - Math.pow(1 - p, 4);
    const val = ease * target;
    const parts = val.toFixed(2).split('.');
    intEl.textContent = fmt(parseFloat(parts[0]));
    if (decEl) decEl.textContent = parts[1];
    if (p < 1) requestAnimationFrame(frame);
  })(performance.now());
}

function showPage(name) {

    if (name === 'budgets')      loadBudgetsPage();
    if (name === 'goals')        loadGoalsPage();

  }
}

async function createAccountFromForm(options = {}) {
  const { showValidation = false } = options;
  const statusEl = document.getElementById('accountCreateStatus');
  const name = document.getElementById('accountName')?.value?.trim();
  const type = document.getElementById('accountType')?.value;
  const initialBalance = document.getElementById('accountBalance')?.value;

async function handleStatementUpload(event) {
  event.preventDefault();
  const statusEl = document.getElementById('statementUploadStatus');
  const submit = document.getElementById('statementUploadSubmit');
  let accountId = document.getElementById('statementAccount')?.value;
  const file = document.getElementById('statementFile')?.files?.[0];
  const statementDateValue = document.getElementById('statementDate')?.value?.trim() || '';
  if (!file) {
    statusEl.textContent = 'Choose a CSV or PDF statement first.';
    statusEl.className = 'form-status err';
    return;
  }
  if (file.size > 50 * 1024 * 1024) {
    statusEl.textContent = 'Statement file is too large. Keep it under 50MB.';
    statusEl.className = 'form-status err';
    return;
  }

  if (!accountId) {
    statusEl.textContent = 'No account selected. Creating one from the form above...';
    statusEl.className = 'form-status';
    let created = null;
    try {
      created = await createAccountFromForm({ showValidation: true });
    } catch (err) {
      statusEl.textContent = err.message || 'Account creation failed.';
      statusEl.className = 'form-status err';
      return;
    }
    if (!created?.id) {
      statusEl.textContent = 'Create an account first, or fill the new account form so upload can create it automatically.';
      statusEl.className = 'form-status err';
      return;
    }
    await loadAccountOptions(true);
    accountId = String(created.id);
    const select = document.getElementById('statementAccount');
    if (select) select.value = accountId;
  }

  const formData = new FormData();
  formData.append('file', file);
  formData.append('accountId', accountId);
  formData.append('source', document.getElementById('statementSource')?.value || 'CSV Upload');
  if (statementDateValue) {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(statementDateValue)) {
      statusEl.textContent = 'Statement date must use the calendar picker or the YYYY-MM-DD format.';
      statusEl.className = 'form-status err';
      return;
    }
    formData.append('statementDate', statementDateValue);
  }
  formData.append('applyToAccountBalance', document.getElementById('statementApplyBalance')?.value || 'false');

  statusEl.textContent = 'Uploading and processing statement...';
  statusEl.className = 'form-status';
  submit.disabled = true;

  let response = null;
  try {
    response = await api('/statements/upload', { method: 'POST', body: formData, throwOnError: true });
  } catch (err) {
    submit.disabled = false;
    statusEl.textContent = err.message === 'Failed to fetch'
      ? 'Upload could not reach the backend. If this PDF is large, try again now that the limit has been increased, or use a smaller/text-based PDF.'
      : (err.message || 'Statement upload failed.');
    statusEl.className = 'form-status err';
    return;
  }
  submit.disabled = false;

  statusEl.textContent = `Imported ${response.importedCount ?? 0} transactions. Skipped ${response.skippedDuplicates ?? 0} duplicates.`;
  statusEl.className = 'form-status ok';
  document.getElementById('statementUploadForm')?.reset();
  await Promise.all([loadStatementHistory(), loadTransactionsPage(0), bootstrap()]);
}

async function loadTransactionsPage(pageNum) {
  txCurrentPage = pageNum;
  const list = document.getElementById('tx-full-list');
  await loadAccountOptions();

  // fetch all once, then filter/paginate client-side
  if (txAllData.length === 0) {

    const raw = await api('/transactions?page=0&size=500');
    if (raw) {
      txAllData = Array.isArray(raw) ? raw : (raw.content ?? raw.transactions ?? []);
    }
    // Also populate category filter
    const cats = [...new Set(txAllData.map(t => t.category ?? t.categoryName).filter(Boolean))];

function renderTxTable() {
  const search  = document.getElementById('tx-search')?.value?.toLowerCase() ?? '';
  const typeF   = document.getElementById('tx-filter-type')?.value ?? '';
  const catF    = document.getElementById('tx-filter-cat')?.value ?? '';

  txFiltered = txAllData.filter(tx => {
    const matchSearch = !search ||
      (tx.description ?? tx.name ?? '').toLowerCase().includes(search) ||
      (tx.category ?? tx.categoryName ?? '').toLowerCase().includes(search);
    const matchType = !typeF ||
      (typeF === 'CREDIT' ? (tx.type ?? '').toUpperCase() === 'INCOME' : (tx.type ?? '').toUpperCase() === 'EXPENSE');
    const matchCat = !catF || (tx.category ?? tx.categoryName) === catF;
    return matchSearch && matchType && matchCat;
  });

  const totalPages = Math.max(1, Math.ceil(txFiltered.length / TX_PAGE_SIZE));
  if (txCurrentPage >= totalPages) txCurrentPage = 0;
  const slice = txFiltered.slice(txCurrentPage * TX_PAGE_SIZE, (txCurrentPage + 1) * TX_PAGE_SIZE);

  const list = document.getElementById('tx-full-list');
  if (slice.length === 0) {
    list.innerHTML = `<div class="page-loading">No transactions found<span class="blink">_</span></div>`;
  } else {
    list.innerHTML = slice.map((tx, i) => {
      const idx    = txCurrentPage * TX_PAGE_SIZE + i + 1;
      const amount = Math.abs(signedAmount(tx));
      const cls   = transactionDisplayClass(tx);
      const sign  = transactionDisplaySign(tx);
      return `
      <div class="tx-full-row">
        <span class="tx-row-num">${idx}</span>
        <div>
          <div class="tx-full-name">${tx.description ?? tx.name ?? '--'}</div>
          <div class="tx-full-sub">${tx.notes ?? ''}</div>
        </div>
        <span class="tx-full-cat">${tx.category ?? tx.categoryName ?? '--'}</span>
        <span class="tx-full-acc">${tx.accountName ?? tx.account ?? '--'}</span>
        <span class="tx-full-date">${relativeDate(tx.date ?? tx.transactionDate)}</span>
        <span class="tx-full-amount ${cls}">${sign}${money(Math.abs(amount))}</span>
      </div>`;
    }).join('');
  }

  document.getElementById('tx-page-info').textContent =
    `PAGE ${txCurrentPage + 1} / ${totalPages} - ${txFiltered.length} TXN`;
  document.getElementById('tx-prev').disabled = txCurrentPage === 0;
  document.getElementById('tx-next').disabled = txCurrentPage >= totalPages - 1;
}

// DONUT RENDERER
// Accepts array of { label, value, color }
// Ã¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?ÂÃ¢?Â
function renderDonut(segments) {
  const svg = document.getElementById('donut');
  // Clear existing circles (keep any injected by previous calls)
  while (svg.firstChild) svg.removeChild(svg.firstChild);

  const cx = 65, cy = 65, r = 48, sw = 10, circ = 2 * Math.PI * r;
  const total = segments.reduce((s, seg) => s + seg.value, 0);

  // Background ring
  const bg = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
  bg.setAttribute('cx', cx); bg.setAttribute('cy', cy);
  bg.setAttribute('r', r - sw / 2 - 8); bg.setAttribute('fill', '#131310');
  svg.appendChild(bg);

  let off = 0;
  segments.forEach(seg => {
    const p = total > 0 ? seg.value / total : 0;
    const el = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    el.setAttribute('cx', cx); el.setAttribute('cy', cy); el.setAttribute('r', r);
    el.setAttribute('fill', 'none'); el.setAttribute('stroke', seg.color); el.setAttribute('stroke-width', sw);
    const dash = p * circ;
    el.setAttribute('stroke-dasharray', `${dash} ${circ - dash}`);
    el.setAttribute('stroke-dashoffset', -off * circ);
    el.style.transform = 'rotate(-90deg)';
    el.style.transformOrigin = `${cx}px ${cy}px`;
    el.style.filter = `drop-shadow(0 0 4px ${seg.color}60)`;
    svg.appendChild(el);
    off += p;
  });
}

function renderBudgets(budgets) {
  const panel = document.querySelector('.bud-row')?.closest('.panel');
  if (!panel) return;

  panel.querySelectorAll('.bud-row, .page-loading').forEach(el => el.remove());

  const fallbackRows = deriveBudgetRowsFromCategories(window.__fintrackDashboardTopCategories || []);
  const sourceRows = (budgets && budgets.length) ? budgets : fallbackRows;

    const title = panel.querySelector('.panel-title');
    title?.insertAdjacentHTML('afterend', `<div class="page-loading" style="padding:18px 0">No budgets configured yet<span class="blink">_</span></div>`);
    return;
  }

  const COLORS = ['var(--acid)', 'var(--ember)', 'var(--ice)', 'var(--offwhite)', '#b06fff'];

  const rows = sourceRows.map((b, i) => {
    const spent  = Number(b.spent ?? b.usedAmount ?? b.totalSpent ?? b.amount ?? 0);
    const limit  = Number((b.limit ?? b.totalAmount ?? b.amount ?? spent) || 1);
    const pct    = Math.min(Math.round(b.percentUsed ?? ((spent / Math.max(limit, 1)) * 100)), 100);
    const rem    = limit - spent;
    const color  = pctColor(pct);
    const statusCls = pct >= 80 ? 'dn' : 'up';
    const statusMsg = pct >= 100
      ? 'LIMIT REACHED'
      : `${pct}% used - ${money(rem)} remaining`;

    return `
    <div class="bud-row">
      <div class="bud-meta">
        <span class="bud-name">${b.name ?? b.category ?? b.categoryName ?? 'Budget'}</span>
        <span class="bud-nums">${money(spent)} / ${money(limit)}</span>
      </div>
      <div class="bud-track">
        <div class="bud-fill" data-w="${pct}" style="background:${color};animation-delay:${i*0.25}s;width:0%"></div>
      </div>
      <div class="bud-status ${statusCls}">${statusMsg}</div>
    </div>`;
  }).join('');

  // Insert before the panel title or at end
  const title = panel.querySelector('.panel-title');
  title ? title.insertAdjacentHTML('afterend', rows) : panel.insertAdjacentHTML('beforeend', rows);

  // Trigger bar animations
  setTimeout(() => {
    panel.querySelectorAll('.bud-fill').forEach(el => { el.style.width = el.dataset.w + '%'; });
  }, 300);
}

function renderAnalyticsChartCompat(data) {
  const svg = document.getElementById('analyticsChartSvg');
  if (!svg) return;

  const W = 820;
  const H = 260;
  const PL = 48;
  const PR = 12;
  const PT = 8;
  const PB = 32;
  const cW = W - PL - PR;
  const cH = H - PT - PB;

  const trend = Array.isArray(data?.monthlyTrend) ? data.monthlyTrend : (Array.isArray(data?.monthlyData) ? data.monthlyData : []);
  let months;
  let income;
  let expenses;

  if (trend.length > 0) {
    const trendMap = new Map();
    trend.forEach(item => {
      const key = normalizeLabel(item.month, '');
      if (!key) return;
      trendMap.set(key, {
        income: Number(item.income || 0) / 1000,
        expenses: Number(item.expenses || 0) / 1000,
      });
    });
    const sortedKeys = [...trendMap.keys()].sort();
    const monthNames = ['JAN', 'FEB', 'MAR', 'APR', 'MAY', 'JUN', 'JUL', 'AUG', 'SEP', 'OCT', 'NOV', 'DEC'];
    const lastKey = sortedKeys[sortedKeys.length - 1];
    const [lastYear, lastMonth] = lastKey.split('-').map(Number);
    const reference = new Date(lastYear, (lastMonth || 1) - 1, 1);
    months = [];
    income = [];
    expenses = [];
    for (let offset = 11; offset >= 0; offset--) {
      const date = new Date(reference.getFullYear(), reference.getMonth() - offset, 1);
      const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}`;
      const point = trendMap.get(key) ?? { income: 0, expenses: 0 };
      months.push(monthNames[date.getMonth()]);
      income.push(point.income);
      expenses.push(point.expenses);
    }
  } else {
    months = ['APR', 'MAY', 'JUN', 'JUL', 'AUG', 'SEP', 'OCT', 'NOV', 'DEC', 'JAN', 'FEB', 'MAR'];
    income = [95, 98, 102, 115, 108, 112, 120, 118, 125, 130, 136, 142];
    expenses = [72, 68, 74, 88, 79, 82, 91, 85, 89, 94, 90, 93.5];
  }

  const net = income.map((val, index) => +(val - expenses[index]).toFixed(1));
  const all = [...income, ...expenses, ...net];
  const minVal = Math.min(0, ...all) * 1.05;
  const maxVal = Math.max(...all, 1) * 1.08;
  const xp = index => PL + (index / Math.max(months.length - 1, 1)) * cW;
  const yp = value => PT + cH - ((value - minVal) / Math.max(maxVal - minVal, 1)) * cH;
  const pathLine = dataset => dataset.map((value, index) => `${index ? 'L' : 'M'}${xp(index).toFixed(1)},${yp(value).toFixed(1)}`).join(' ');
  const pathArea = dataset => {
    const baseline = PT + cH;
    return `${pathLine(dataset)} L${xp(dataset.length - 1).toFixed(1)},${baseline} L${xp(0).toFixed(1)},${baseline} Z`;
  };
  const last = months.length - 1;

  svg.innerHTML = `<defs>
      <linearGradient id="agA" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stop-color="#c8f000" stop-opacity="0.14"/><stop offset="100%" stop-color="#c8f000" stop-opacity="0"/></linearGradient>
      <linearGradient id="agB" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stop-color="#ff4d00" stop-opacity="0.10"/><stop offset="100%" stop-color="#ff4d00" stop-opacity="0"/></linearGradient>
      <filter id="agf"><feGaussianBlur stdDeviation="2" result="b"/><feMerge><feMergeNode in="b"/><feMergeNode in="SourceGraphic"/></feMerge></filter>
    </defs>
    ${[0, 0.25, 0.5, 0.75, 1].map(tick => {
      const y = PT + tick * cH;
      const value = maxVal - tick * (maxVal - minVal);
      return `<line x1="${PL}" y1="${y.toFixed(1)}" x2="${W - PR}" y2="${y.toFixed(1)}" stroke="rgba(255,255,255,0.04)" stroke-width="1"/><text x="${PL - 6}" y="${(y + 3).toFixed(1)}" text-anchor="end" font-family="IBM Plex Mono" font-size="8" fill="#3a3a36">${value.toFixed(0)}K</text>`;
    }).join('')}
    ${months.map((month, index) => `<text x="${xp(index).toFixed(1)}" y="${H - 5}" text-anchor="middle" font-family="IBM Plex Mono" font-size="8" fill="#3a3a36">${month}</text>`).join('')}
    <path d="${pathArea(income)}" fill="url(#agA)"/>
    <path d="${pathArea(expenses)}" fill="url(#agB)"/>
    <path d="${pathLine(net)}" fill="none" stroke="rgba(232,228,220,0.2)" stroke-width="1" stroke-dasharray="3 3"/>
    <path id="aeL" d="${pathLine(expenses)}" fill="none" stroke="#ff4d00" stroke-width="1.5" filter="url(#agf)"/>
    <path id="aiL" d="${pathLine(income)}" fill="none" stroke="#c8f000" stroke-width="2" filter="url(#agf)"/>
    <circle cx="${xp(last).toFixed(1)}" cy="${yp(income[last]).toFixed(1)}" r="4" fill="#c8f000" style="filter:drop-shadow(0 0 6px #c8f000)"/>
    <circle cx="${xp(last).toFixed(1)}" cy="${yp(expenses[last]).toFixed(1)}" r="3" fill="#ff4d00" style="filter:drop-shadow(0 0 6px #ff4d00)"/>`;

  ['aiL', 'aeL'].forEach((id, index) => {
    const el = svg.querySelector(`#${id}`);
    if (!el || typeof el.getTotalLength !== 'function') return;
    const length = el.getTotalLength();
    el.style.strokeDasharray = length;
    el.style.strokeDashoffset = length;
    setTimeout(() => {
      el.style.transition = 'stroke-dashoffset 2s cubic-bezier(0.22,1,0.36,1)';
      el.style.strokeDashoffset = '0';
    }, 300 + index * 240);
  });
}

  const insightSavingsPotential = document.getElementById('insightSavingsPotential');
  const insightRecommendationsList = document.getElementById('insightRecommendationsList');

  if (insightPredictionVal) {
    insightPredictionVal.textContent = insights?.predictedNextMonthExpense != null
      ? money(insights.predictedNextMonthExpense)
      : '--';
  }
  if (insightSavingsPotential) {
    insightSavingsPotential.textContent = insights?.savingsPotential != null
      ? money(insights.savingsPotential)
      : '--';
  }
  if (insightPatternsList) {
    const patterns = Array.isArray(insights?.patterns) ? insights.patterns : [];
    insightPatternsList.innerHTML = patterns.length
      ? patterns.map(item => `<div class="insight-mini"><strong>${item.title ?? 'Pattern'}:</strong> ${item.detail ?? ''}</div>`).join('')
      : `<div class="page-loading" style="padding:12px 0">No behavior patterns yet</div>`;
  }
  if (insightAnomalyCount) {
    const anomalies = Array.isArray(insights?.anomalies) ? insights.anomalies : [];
    insightAnomalyCount.textContent = String(anomalies.length);
    if (insightAnomaliesList) {
      insightAnomaliesList.innerHTML = anomalies.length
        ? anomalies.map(item => `<div class="insight-mini"><strong>${item.description ?? 'Transaction'}:</strong> ${item.reason ?? 'Unusual activity detected'}</div>`).join('')
        : `<div class="page-loading" style="padding:12px 0">No anomalies detected</div>`;
    }
  }
  if (insightRecommendationsList) {
    const recommendations = Array.isArray(insights?.recommendations) ? insights.recommendations : [];
    insightRecommendationsList.innerHTML = recommendations.length
      ? recommendations.map(item => `<div class="insight-mini"><strong>${item.title ?? 'Recommendation'}:</strong> ${item.detail ?? item.action ?? ''}</div>`).join('')
      : `<div class="page-loading" style="padding:12px 0">No recommendations yet</div>`;
  }
}

async function bootstrap() {
  if (!authState.accessToken) {
    setConnectionState('Demo Mode', 'var(--ember)');
    showAuthModal('login');
    return;
  }

  // Kick off all requests at once Ã¢?? no waterfalling
  const [dashboard, transactions, budgets, goals, accounts, insights, financialProfile] = await Promise.all([
    api('/analytics/dashboard'),
    api('/transactions?page=0&size=8'),
    api('/budgets'),
    api('/savings-goals'),
    api('/accounts'),
    api('/insights'),
    api('/goals'),
  ]);

  window.__fintrackDashboardTopCategories = Array.isArray(dashboard?.topCategories) ? dashboard.topCategories : [];
  window.__fintrackFinancialProfile = financialProfile ?? null;
  window.__fintrackCurrentExpenses = Number(dashboard?.totalExpenses ?? dashboard?.monthlyExpenses ?? dashboard?.currentMonthExpenses ?? 0);

  // Apply data to each section
  applyDashboard(dashboard);
  renderChart(dashboard);
  renderTransactions(
    // API might return { content: [...] } (Page) or plain array
    Array.isArray(transactions) ? transactions :
    transactions?.content ?? transactions?.transactions ?? null
  );
  renderBudgets(
    Array.isArray(budgets) ? budgets :
    budgets?.content ?? budgets?.budgets ?? null
  );
  renderGoals(
    Array.isArray(goals) ? goals :
    goals?.content ?? goals?.savingsGoals ?? null
  );
  renderOverviewInsights(insights, dashboard);
  renderAccounts(
    Array.isArray(accounts) ? accounts :
    accounts?.content ?? accounts?.accounts ?? null
  );

  // Animate bars that may have been rendered before bootstrap
  setTimeout(() => {
    document.querySelectorAll('.bud-fill').forEach(el => { if (!el.style.width || el.style.width === '0%') el.style.width = el.dataset.w + '%'; });
    document.querySelectorAll('.goal-bar-fill').forEach(el => { if (!el.style.width || el.style.width === '0%') el.style.width = el.dataset.w + '%'; });
  }, 600);

  const connected = !!(dashboard ?? transactions ?? accounts);
  setConnectionState(connected ? 'Live' : 'Offline', connected ? 'var(--acid)' : 'var(--ember)');

  if (!connected) showAuthModal('login');
}

async function handleAuthSubmit(event) {
  event.preventDefault();
  const submit = document.getElementById('authSubmit');
  const email = document.getElementById('authEmail')?.value?.trim();
  const password = document.getElementById('authPassword')?.value || '';
  const fullName = document.getElementById('authFullName')?.value?.trim();

  if (!email || !password || (authMode === 'register' && !fullName)) {
    setAuthError('Complete the required fields first.');
    return;
  }

  setAuthError('');
  if (submit) {
    submit.disabled = true;
    submit.textContent = authMode === 'register' ? 'Creating...' : 'Signing In...';
  }

  try {
    const endpoint = authMode === 'register' ? '/auth/register' : '/auth/login';
    const payload = authMode === 'register'
      ? { email, password, fullName, currency: 'INR' }
      : { email, password };
    const next = await authRequest(endpoint, payload);
    saveAuthState({
      accessToken: next.accessToken,
      refreshToken: next.refreshToken,
      user: next.user,
    });
    hideAuthModal();
    await bootstrap();
  } catch (err) {
    const message = err.message === 'Email already in use'
      ? 'This email is already registered. Switch to Login and use your password.'
      : (err.message || 'Authentication failed.');
    setAuthError(message);
    showAuthModal(authMode);
  } finally {
    if (submit) submit.disabled = false;
    setAuthMode(authMode);
  }
}

async function handleLogout() {
  await api('/auth/logout', { method: 'POST', retryAuth: false });
  clearAuthState();
  showAuthModal('login');
}

function bindAuthUi() {
  document.getElementById('userChip')?.addEventListener('click', () => showAuthModal(authState.accessToken ? authMode : 'login'));
  document.getElementById('authClose')?.addEventListener('click', () => hideAuthModal());
  document.getElementById('authForm')?.addEventListener('submit', handleAuthSubmit);
  document.getElementById('authRefresh')?.addEventListener('click', async () => {
    hideAuthModal();
    await bootstrap();
  });
  document.getElementById('authLogout')?.addEventListener('click', handleLogout);
  document.querySelectorAll('[data-auth-mode]').forEach(el => {
    el.addEventListener('click', () => setAuthMode(el.dataset.authMode));
  });
}
